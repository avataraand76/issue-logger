CREATE TABLE tb_category (
    id_category BIGINT PRIMARY KEY AUTO_INCREMENT,
    name_category TEXT NOT NULL
);

CREATE TABLE tb_issue (
    id_issue BIGINT PRIMARY KEY AUTO_INCREMENT,
    id_category BIGINT,
    name_issue TEXT NOT NULL
);

CREATE TABLE tb_solution (
    id_solution BIGINT PRIMARY KEY AUTO_INCREMENT,
    id_category BIGINT,
    name_solution TEXT NOT NULL
);

CREATE TABLE tb_machinery_category (
    id_machinery_category BIGINT PRIMARY KEY AUTO_INCREMENT,
    name_machinery_category TEXT NOT NULL
);

CREATE TABLE tb_machinery (
    id_machinery BIGINT PRIMARY KEY AUTO_INCREMENT,
    id_machinery_category BIGINT,
    code_machinery TEXT NOT NULL,
    name_machinery TEXT NOT NULL
);

CREATE TABLE tb_workshop (
    id_workshop BIGINT PRIMARY KEY AUTO_INCREMENT,
    name_workshop TEXT NOT NULL
);

CREATE TABLE tb_line (
    id_line BIGINT PRIMARY KEY AUTO_INCREMENT,
    id_workshop BIGINT,
    name_line TEXT NOT NULL
);

CREATE TABLE tb_employee_group (
    id_employee_group BIGINT PRIMARY KEY AUTO_INCREMENT,
    name_employee_group TEXT NOT NULL
);

CREATE TABLE tb_employee (
    id_employee BIGINT PRIMARY KEY AUTO_INCREMENT,
    id_workshop BIGINT,
	id_line BIGINT,
    id_employee_group BIGINT,
    full_name TEXT NOT NULL,
    position TEXT NOT NULL
);

CREATE TABLE tb_logged_issue (
    id_logged_issue BIGINT PRIMARY KEY AUTO_INCREMENT,
    submission_time DATETIME NOT NULL,
    line_number TEXT NOT NULL,
    station_number TEXT NOT NULL,
    id_category BIGINT,
    machinery_type TEXT,
    machinery_code TEXT,
    issue_description TEXT,
    solution_description TEXT,
    problem_solver TEXT,
    responsible_person TEXT NOT NULL,
    end_time DATETIME,
    downtime_minutes INT,
    old_product_code TEXT,
    new_product_code TEXT,
    workshop TEXT NOT NULL,
	factory TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
	status_logged_issue ENUM('pending', 'resolved') DEFAULT 'pending'
);
